\begin{abstract}
This supplement gives complete proofs for the narrow Paper-I model: an arbitrary nonempty binary
block code, exactly one uniformly located deletion, and independent BSC substitutions. It proves the
aggregate likelihood, strict best-history/aggregate-ML reversal, inverse-alignment discovery-shell
characterization, exact shell certificate, realization-dependent stopping theorem, finite-length
work bounds, and high-probability membership/query exponent. It also records boundary cases and the
claims that are intentionally not made.
\end{abstract}

\section{Model and exact likelihood}
Let $X=(X_1,\ldots,X_n)\in\{0,1\}^n$. Independent substitutions form
$\widetilde X_i=X_i\oplus N_i$ with $N_i\sim\mathrm{Bernoulli}(p_s)$, $0\le p_s<1/2$.
Exactly one coordinate $J$ is deleted uniformly from $\{1,\ldots,n\}$. The observation
$Y\in\{0,1\}^{m}$ has $m=n-1$. The code $\C\subseteq\{0,1\}^n$ is arbitrary and nonempty,
with uniform prior, and is accessed through a membership test. Define
\[
 d_j(x,y)=\sum_{i=1}^{j-1}\one\{x_i\ne y_i\}+
 \sum_{i=j+1}^{n}\one\{x_i\ne y_{i-1}\}.
\]
\begin{proposition}[Likelihood and recurrence]\label{prop:like}
For every $x,y$,
\[
 W(y\mid x)=\frac1n\sum_{j=1}^{n}p_s^{d_j(x,y)}(1-p_s)^{m-d_j(x,y)}.
\]
Moreover,
\[
 d_1=\sum_{i=2}^{n}\one\{x_i\ne y_{i-1}\},\qquad
 d_{j+1}=d_j-\one\{x_{j+1}\ne y_j\}+\one\{x_j\ne y_j\}.
\]
Thus all alignment components are evaluated in $O(n)$ arithmetic operations.
\end{proposition}
\begin{proof}
Condition on $J=j$. The deleted symbol is marginalized, and the $m$ surviving comparisons require
exactly $d_j$ substitutions. Independence gives the $j$th term; averaging over the uniform deletion
coordinate gives the likelihood. Moving the hypothesized deletion from $j$ to $j+1$ changes only
the comparison involving $y_j$: the old suffix term $(x_{j+1},y_j)$ is removed and the new prefix
term $(x_j,y_j)$ is added. This proves the recurrence and complexity statement.
\end{proof}

\section{Strict pathwise failure}
Let
\[
 M_y(x)=\max_j\frac1n p_s^{d_j(x,y)}(1-p_s)^{m-d_j(x,y)}.
\]
\begin{theorem}[Strict reversal]\label{thm:rev}
For $n\ge4$ and $0<p_s<1/2$, set
\[
 y=0^{n-3}10,\qquad x_A=0^{n-3}101,\qquad x_B=0^n,\qquad \C=\{x_A,x_B\}.
\]
Then $M_y(x_A)>M_y(x_B)$, whereas
\[
 W(y\mid x_B)>W(y\mid x_A)\quad\Longleftrightarrow\quad np_s>1+2p_s^2.
\]
\end{theorem}
\begin{proof}
Deleting the last coordinate of $x_A$ gives $y$ without substitution, so
$M_y(x_A)=n^{-1}(1-p_s)^m$. Every deletion of $x_B$ leaves $0^m$, which differs from $y$ once,
so $M_y(x_B)=n^{-1}p_s(1-p_s)^{m-1}$; the first inequality follows from $p_s<1/2$.
For $x_A$, mismatch distances $0,1,2,3$ occur with multiplicities $1,1,1,n-3$; every alignment
of $x_B$ has distance one. Therefore
\begin{align*}
 W(y\mid x_B)&=p_s(1-p_s)^{n-2},\\
 W(y\mid x_A)&=\frac1n\bigl[(1-p_s)^{n-1}+p_s(1-p_s)^{n-2}
 +p_s^2(1-p_s)^{n-3}+(n-3)p_s^3(1-p_s)^{n-4}\bigr].
\end{align*}
Direct factorization yields
\[
 \frac{n(W(y\mid x_B)-W(y\mid x_A))}{(1-p_s)^{n-4}}
 =(1-2p_s)(np_s-1-2p_s^2).
\]
The first factor is positive, proving the equivalence.
\end{proof}

\section{Discovery shells}
For $z\in\{0,1\}^m$, let $\I(z)$ be the set of binary length-$n$ words formed by one insertion.
\begin{lemma}[One-insertion sphere]\label{lem:sphere}
For every $z$, $|\I(z)|=m+2=n+1$. For inserted bit $b$, all distinct outputs arise by inserting
$b$ in the first gap and immediately after each symbol of $z$ unequal to $b$.
\end{lemma}
\begin{proof}
Moving $b$ through a run of $b$ does not change the output. A new output appears initially and each
time the insertion crosses a symbol unequal to $b$, giving $1+\#\{i:z_i\ne b\}$. Summing over
$b=0,1$ gives $m+2$. The two output classes have different Hamming weights and are disjoint.
\end{proof}
\begin{remark}
Lemma~\ref{lem:sphere} is classical and is used only to remove redundant history-pair generation.
It is not asserted as a new coding theorem.
\end{remark}
Define the discovery shell $h_y(x)=\min_jd_j(x,y)$.
\begin{proposition}[Shell characterization]\label{prop:shell}
When shell $s$ enumerates all error patterns $e$ of weight $s$, sets $z=y\oplus e$, and generates
$\I(z)$, candidate $x$ is generated by the end of shell $s$ iff $h_y(x)\le s$. Its first shell is
$h_y(x)$.
\end{proposition}
\begin{proof}
If alignment $j$ is hypothesized, deleting $x_j$ gives $z=y\oplus e$ with
$\wt(e)=d_j(x,y)$; reinserting $x_j$ at $j$ reconstructs $x$. Conversely, every generated word
comes with such an alignment and error vector. Minimization over $j$ proves both directions.
\end{proof}

\section{Certified exact decoder}
For $0<p_s<1/2$, after completing shell $s<m$, define
\[
 U_s=p_s^{s+1}(1-p_s)^{m-s-1}.
\]
\begin{theorem}[Unseen-candidate certificate]\label{thm:cert}
Every candidate unseen after shell $s<m$ has likelihood at most $U_s$. If a completely scored
incumbent codeword has likelihood strictly greater than $U_s$, the current incumbent set is the
complete global ML tie set. Exhaustion at shell $m$ always returns the exact tie set.
\end{theorem}
\begin{proof}
By Proposition~\ref{prop:shell}, unseen $x$ satisfies $d_j(x,y)\ge s+1$ for every $j$. Since
$p_s/(1-p_s)<1$, every term in the likelihood is at most
$p_s^{s+1}(1-p_s)^{m-s-1}$; averaging does not change that bound. Strict incumbent comparison
excludes both an unseen maximizer and an unseen tie. At shell $m$, all $2^n$ candidates have been
generated, so no unseen candidate remains.
\end{proof}
At $p_s=0$, only shell zero has positive probability, and shell zero contains every positive-
likelihood candidate. At $p_s=1/2$, shell ordering provides no strict decay; exhaustive finite
termination remains valid, but the strict-gain results below do not apply.

\section{Realization-dependent stopping}
Let $T$ be the number of substitutions among the $m$ surviving symbols, let
$\rho=p_s/(1-p_s)$, and define
\[
 L_n(p_s)=\min\{\ell\ge1:\rho^\ell<1/n\}
 =\left\lfloor\log_{(1-p_s)/p_s}n\right\rfloor+1.
\]
\begin{theorem}[Stopping shell]\label{thm:stop}
For $0<p_s<1/2$, the exact decoder stops no later than
\[
 S\le\min\{m,T+L_n(p_s)-1\}.
\]
This bound is independent of the code.
\end{theorem}
\begin{proof}
Under the true deletion alignment, the transmitted word has mismatch count $T$, so it is discovered
by shell $T$ and contributes at least $n^{-1}p_s^T(1-p_s)^{m-T}$. The incumbent is at least this
large. At $s=T+L_n-1$,
\[
 \frac{U_s}{n^{-1}p_s^T(1-p_s)^{m-T}}=n\rho^{L_n}<1,
\]
so Theorem~\ref{thm:cert} certifies by this shell unless finite exhaustion occurs earlier.
\end{proof}

Let $\Vol(m,s)=\sum_{w=0}^{\min\{m,s\}}\binom mw$.
\begin{corollary}[Finite work]\label{cor:work}
For the realized $T$, with $S$ as in Theorem~\ref{thm:stop},
\[
 Q_{\rm gen},Q_{\rm mem}\le(n+1)\Vol(m,S),\qquad Q_{\rm score}\le Q_{\rm mem}.
\]
A history-pair generator attempts $2n\Vol(m,S)$ insertion pairs before global deduplication.
\end{corollary}
\begin{proof}
Each error vector contributes exactly $n+1$ distinct one-insertion supersequences. A global seen set
can only reduce membership calls, and only words passing membership receive a complete likelihood
evaluation.
\end{proof}

If $\tau_\delta$ is the smallest binomial quantile satisfying $\Prb[T\le\tau_\delta]\ge1-\delta$,
then with probability at least $1-\delta$,
\[
 Q_{\rm mem}\le(n+1)\Vol\bigl(m,\min\{m,\tau_\delta+L_n-1\}\bigr).
\]
Taking expectation over $T\sim\mathrm{Binomial}(m,p_s)$ gives
\[
 \mathbb E Q_{\rm mem}\le(n+1)\sum_{t=0}^{m}\binom mtp_s^t(1-p_s)^{m-t}
 \Vol\bigl(m,\min\{m,t+L_n-1\}\bigr).
\]

\section{High-probability query exponent}
Let $h_2(u)=-u\log_2u-(1-u)\log_2(1-u)$.
\begin{theorem}[Oracle/query exponent]\label{thm:exp}
Fix $0<p_s<1/2$ and $c>0$. As $n\to\infty$, with probability at least $1-n^{-c}$,
\[
 Q_{\rm mem},Q_{\rm score}\le2^{n(h_2(p_s)+o(1))}.
\]
If $|\C_n|=2^{nR+o(n)}$ with $R>h_2(p_s)$, then exhaustive codeword-wise ML performs at least
$2^{n(R-h_2(p_s)-o(1))}$ times as many complete candidate-likelihood evaluations.
\end{theorem}
\begin{proof}
Put $\epsilon_n=\sqrt{c\ln n/(2m)}$. Hoeffding's inequality gives
$\Prb[T>m(p_s+\epsilon_n)]\le n^{-c}$. Also $L_n=O(\log n)$, so on the complementary event
$S/m\le p_s+o(1)<1/2$. The standard Hamming-ball bound gives
$\Vol(m,S)\le2^{mh_2(S/m)}=2^{n(h_2(p_s)+o(1))}$. The polynomial factor $n+1$ is absorbed into
$2^{o(n)}$, and Corollary~\ref{cor:work} proves the first statement. Division of
$|\C_n|$ by $Q_{\rm score}$ proves the second.
\end{proof}

\section{Claim boundaries}
The proofs establish exact aggregate ML, a strict pathwise counterexample, a code-independent
stopping shell, and membership/query bounds for the stated one-deletion model. They do not show
instance optimality, universal latency superiority, multiple-edit performance, or novelty of
path summation, threshold aggregation, GRAND-style membership testing, or insertion-sphere
cardinality. The finite campaign contained no naturally sampled case in which the first-shell
codeword tie set was disjoint from the aggregate-ML tie set; the strict structural claim rests on
Theorem~\ref{thm:rev}, not on empirical prevalence.
